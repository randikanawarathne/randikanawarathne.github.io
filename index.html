from flask import Flask, jsonify, request
from flask_cors import CORS
import json

app = Flask(__name__)
CORS(app)

# Sample data
users = [
    {"id": 1, "name": "John Doe", "email": "john@example.com"},
    {"id": 2, "name": "Jane Smith", "email": "jane@example.com"}
]

@app.route('/')
def home():
    return jsonify({"message": "Welcome to Python API"})

@app.route('/users', methods=['GET'])
def get_users():
    return jsonify(users)

@app.route('/users/', methods=['GET'])
def get_user(user_id):
    user = next((u for u in users if u['id'] == user_id), None)
    if user:
        return jsonify(user)
    return jsonify({"error": "User not found"}), 404

@app.route('/users', methods=['POST'])
def add_user():
    new_user = request.get_json()
    new_user['id'] = len(users) + 1
    users.append(new_user)
    return jsonify(new_user), 201

if __name__ == '__main__':
    app.run(debug=True, port=5000)

// Insert project
function addProject($title, $description, $technologies) {
    global $conn;
    $sql = "INSERT INTO projects (title, description, technologies) 
            VALUES (:title, :description, :technologies)";
    $stmt = $conn->prepare($sql);
    return $stmt->execute([
        ':title' => $title,
        ':description' => $description,
        ':technologies' => $technologies
    ]);
}

// Get all projects
function getProjects() {
    global $conn;
    $stmt = $conn->query("SELECT * FROM projects ORDER BY created_at DESC");
    return $stmt->fetchAll(PDO::FETCH_ASSOC);
}

// Update project
function updateProject($id, $title, $description, $technologies) {
    global $conn;
    $sql = "UPDATE projects SET 
            title = :title,
            description = :description,
            technologies = :technologies
            WHERE id = :id";
    $stmt = $conn->prepare($sql);
    return $stmt->execute([
        ':id' => $id,
        ':title' => $title,
        ':description' => $description,
        ':technologies' => $technologies
    ]);
}

// Delete project
function deleteProject($id) {
    global $conn;
    $sql = "DELETE FROM projects WHERE id = :id";
    $stmt = $conn->prepare($sql);
    return $stmt->execute([':id' => $id]);
}

JavaScript Weather App


class WeatherApp {
    constructor(apiKey) {
        this.apiKey = apiKey;
        this.baseUrl = 'https://api.openweathermap.org/data/2.5/weather';
        this.init();
    }


    init() {
        this.bindEvents();
        this.loadWeather('Colombo');
    }

    bindEvents() {
        document.getElementById('search-btn').addEventListener('click', () => {
            const city = document.getElementById('city-input').value;
            if (city) {
                this.loadWeather(city);
            }
        });

        document.getElementById('city-input').addEventListener('keypress', (e) => {
            if (e.key === 'Enter') {
                const city = e.target.value;
                if (city) {
                    this.loadWeather(city);
                }
            }
        });
    }

    async loadWeather(city) {
        try {
            const url = `${this.baseUrl}?q=${city}&appid=${this.apiKey}&units=metric`;
            const response = await fetch(url);
            const data = await response.json();

            if (data.cod === 200) {
                this.displayWeather(data);
            } else {
                this.showError('City not found');
            }
        } catch (error) {
            this.showError('Failed to fetch weather data');
        }
    }

    displayWeather(data) {
        const weatherInfo = document.getElementById('weather-info');
        weatherInfo.innerHTML = `
            

                
${data.name}, ${data.sys.country}

                

                    
                    
${Math.round(data.main.temp)}°C

                

                
${data.weather[0].description}

                

                    

                        Humidity
                        ${data.main.humidity}%
                    

                    

                        Wind
                        ${data.wind.speed} m/s
                    

                    

                        Pressure
                        ${data.main.pressure} hPa
                    

                

            

        `;
    }

    showError(message) {
        const weatherInfo = document.getElementById('weather-info');
        weatherInfo.innerHTML = `
            

                
                
${message}

            

        `;
    }
}

// Initialize app
document.addEventListener('DOMContentLoaded', () => {

});
